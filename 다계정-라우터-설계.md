interface AccountPoolRule {
  id: string;
  userId: string | null;   // null이면 전체 적용
  provider: string | null; // null이면 모든 제공자
  modelPattern: string;    // 예: "claude-3-*", "gpt-4*"
  
  allowedAccountIds: string[];  // 사용 가능한 계정 목록
  selectionStrategy: 'round_robin' | 'priority' | 'least_used';
  enableFallback: boolean;
}

**계정 선택 로직:**
```typescript
class AccountPoolManager {
  async selectAccount(
    userId: string,
    model: string,
    provider: string
  ): Promise<UpstreamAccount> {
    // 1. 규칙 조회
    const rule = await this.findRule(userId, provider, model);
    
    // 2. 사용 가능한 계정 필터링
    const candidates = await this.filterAccounts({
      accountIds: rule.allowedAccountIds,
      status: 'active',
      hasQuotaRemaining: true
    });
    
    if (candidates.length === 0) {
      if (rule.enableFallback) {
        // 다른 제공자로 폴백
        return this.selectFallbackAccount(model);
      }
      throw new Error('No available accounts');
    }
    
    // 3. 전략에 따라 선택
    switch (rule.selectionStrategy) {
      case 'round_robin':
        return this.roundRobinSelect(candidates);
      case 'priority':
        return candidates.sort((a, b) => a.priority - b.priority)[0];
      case 'least_used':
        return candidates.sort((a, b) => a.usedToday - b.usedToday)[0];
    }
  }
  
  // 계정 상태 업데이트
  async updateAccountStatus(
    accountId: string,
    result: { success: boolean; error?: Error; tokensUsed?: number }
  ) {
    if (result.success) {
      await this.incrementUsage(accountId, result.tokensUsed);
    } else {
      // 429 에러면 할당량 초과로 표시
      if (result.error?.message.includes('429')) {
        await this.markQuotaExceeded(accountId);
      }
      // 401/403이면 자격증명 문제
      else if (result.error?.message.match(/40[13]/)) {
        await this.markError(accountId, 'Invalid credentials');
      }
    }
  }
}
```

interface Decoder {

class OpenAIResponsesDecoder implements Decoder { /* ... */ }

// 타입 IR (중간 표현)
interface ParsedRequest {
  systemPrompt: string[];
  messages: Message[];
  model: string;
  temperature?: number;
  maxTokens?: number;
  tools?: Tool[];
  providerState?: Record<string, unknown>;  // 제공자별 임시 상태

  role: 'user' | 'assistant' | 'developer' | 'toolResult';

}

type MessagePart = 
  | { type: 'text'; text: string }
  | { type: 'thinking'; text: string; signature?: string; redacted?: string[] }
  | { type: 'toolCall'; id: string; name: string; args: unknown; thoughtSignature?: string };

// 출력 어댑터 (M개)
interface Adapter {
  buildRequest(parsed: ParsedRequest): ProviderRequest;
  parseStream(response: ReadableStream): AsyncIterable<StreamEvent>;
}

// 통합 이벤트 타입
type StreamEvent =
  | { type: 'heartbeat' }
  | { type: 'text_delta'; text: string; phase?: 'commentary' | 'final_answer' }
  | { type: 'thinking_delta'; text: string }
  | { type: 'thinking_signature'; signature: string }
  | { type: 'redacted_thinking'; markers: string[] }
  | { type: 'tool_call_start'; id: string; name: string }
  | { type: 'tool_call_delta'; id: string; argsDelta: string }
  | { type: 'tool_call_end'; id: string; args: unknown }
  | { type: 'done'; finishReason: string; usage: TokenUsage }
  | { type: 'incomplete'; reason: string }
  | { type: 'error'; error: Error };

// 출력 인코더 (3개)
interface Encoder {
  encode(events: AsyncIterable<StreamEvent>): AsyncIterable<string>;
}

class ResponsesEncoder implements Encoder {
  // "열린 항목 하나" 상태기계
  private currentItem: {
    type: 'reasoning' | 'message' | 'function_call' | null;
    index: number;
  } = { type: null, index: 0 };
  
  async *encode(events: AsyncIterable<StreamEvent>) {
    for await (const event of events) {
      if (this.shouldCloseCurrentItem(event)) {
        yield* this.closeCurrentItem();
      }
      yield* this.handleEvent(event);
    }
  }
}

### 2.4 재전송 게이트 (보고서 섹션 11)

```typescript
class ResendGate {
  private firstByteReceived = false;
  
  canRetry(): boolean {
    // 첫 출력 바이트 이전에만 재시도 허용
    return !this.firstByteReceived;
  }
  
  markFirstByte() {
    this.firstByteReceived = true;
  }
  
  async executeWithRetry<T>(
    fn: () => Promise<T>,
    accountSelector: () => Promise<UpstreamAccount>
  ): Promise<T> {
    let attempts = 0;
    const maxAttempts = 3;
    
    while (attempts < maxAttempts) {
      const account = await accountSelector();
      
      try {
        const result = await fn();
        
        // 스트림이 시작되면 더 이상 재시도 불가
        if (this.firstByteReceived) {
          return result;
        }
        
      } catch (error) {
        attempts++;
        
        // 첫 바이트 이후 오류 → incomplete로 종료
        if (this.firstByteReceived) {
          throw new Error('Stream interrupted', { cause: error });
        }
        
        // 재시도 가능한 오류인지 확인
        if (error.status === 429 || error.status === 503) {
          await this.markAccountUnavailable(account);
          continue;  // 다른 계정으로 재시도
        }
        
        throw error;
      }
    }
    
    throw new Error('Max retry attempts exceeded');
  }
}
```

### 5.1 자격증명 보호

```typescript
// AES-256-GCM 암호화
import crypto from 'crypto';

class CredentialVault {
  private key: Buffer;  // 환경변수 ENCRYPTION_KEY에서 로드
  
  encrypt(plaintext: string): string {
    const iv = crypto.randomBytes(16);
    const cipher = crypto.createCipheriv('aes-256-gcm', this.key, iv);
    
    let encrypted = cipher.update(plaintext, 'utf8', 'hex');
    encrypted += cipher.final('hex');
    
    const authTag = cipher.getAuthTag();
    
    // iv:authTag:ciphertext 형식
    return `${iv.toString('hex')}:${authTag.toString('hex')}:${encrypted}`;
  }
  
  decrypt(ciphertext: string): string {
    const [ivHex, authTagHex, encrypted] = ciphertext.split(':');
    
    const iv = Buffer.from(ivHex, 'hex');
    const authTag = Buffer.from(authTagHex, 'hex');
    const decipher = crypto.createDecipheriv('aes-256-gcm', this.key, iv);
    decipher.setAuthTag(authTag);
    
    let plaintext = decipher.update(encrypted, 'hex', 'utf8');
    plaintext += decipher.final('utf8');
    
    return plaintext;
  }
}
```

## 9. opencodex vs OmniRoute 차이 요약

| 항목 | opencodex | OmniRoute | 이 설계 |
|------|-----------|-----------|---------|
| **주 용도** | Codex/Claude Code 전용 | 범용 게이트웨이 | **다계정 라우팅** |
| **프로토콜 정확도** | ⭐⭐⭐⭐⭐ | ⭐⭐⭐ | ⭐⭐⭐⭐ (opencodex IR 기반) |
| **제공자 수** | ~10개 | 375개 | 주요 3-5개 (확장 가능) |
| **다계정 지원** | ❌ 단일 계정 | ✅ 키 목록 (무작위) | ✅ **계정 풀 + 전략** |
| **사용자 격리** | ❌ | ❌ (REQUIRE_API_KEY=false) | ✅ **사용자별 인증** |
| **할당량 추적** | 간단 | 고급 (대시보드) | **계정별 할당량** |
| **코드 크기** | ~4만 줄 | ~15만 줄 | 목표 ~1만 줄 |

## 10. 시작하기

```bash
# 1. 프로젝트 생성
mkdir multi-account-router
cd multi-account-router
npm init -y

# 2. 의존성 설치
npm install express typescript @types/express
npm install pg @types/pg                    # PostgreSQL
npm install express-rate-limit
npm install dotenv zod                       # 설정 검증
npm install pino pino-pretty                 # 로깅
npm install prom-client                      # 메트릭

# 3. 개발 의존성
npm install -D tsx nodemon @types/node

# 4. TypeScript 설정
npx tsc --init

# 5. 데이터베이스 초기화
npm install -g db-migrate db-migrate-pg
db-migrate create initial-schema --sql-file

# 6. 실행
npm run dev
```
